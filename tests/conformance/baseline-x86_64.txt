# Known non-PASS conformance cases on x86_64 (IR0 dev 629036ef).
# Format: <case> <PASS|FAIL|TCONF|BROK>  # why. Unlisted cases must PASS.
# A case that starts passing is reported as FIXED: remove its line.
# See Documentation/conformance.md.

fs_unlink_open_file FAIL  # an unlinked file cannot be read through an fd still open on it (path-based VFS)
mm_madvise_dontneed TCONF  # madvise is ENOSYS
time_clock_getres TCONF  # clock_getres is ENOSYS
procfs_cpu_jiffies_vs_uptime FAIL  # /proc/stat cpu prints scheduler ticks (1000 Hz), not USER_HZ
procfs_task_cpu_times FAIL  # utime/stime/starttime in ticks; times(2) and getrusage(2) are ENOSYS
procfs_stat_processes_forks FAIL  # "processes" counts live tasks, not forks since boot
procfs_swap_consistent FAIL  # SwapTotal/SwapFree missing from /proc/meminfo
procfs_loadavg_runqueue FAIL  # procs_running and loadavg count only the task on the CPU; integer EMA stalls
sysfs_cpu_lists FAIL  # /sys/devices/system/cpu/{online,possible,present} missing
procfs_pid_stat_fields FAIL  # /proc/<pid>/stat has 36 of 52 fields; nice fixed at 0
procfs_pid_status_keys FAIL  # status lacks Tgid, Rss* and ctxt_switches
procfs_task_dir FAIL  # /proc/<pid>/task missing
procfs_sys_kernel_vmstat FAIL  # /proc/sys/kernel/{pid_max,osrelease} and /proc/vmstat missing
procfs_cpuinfo_format FAIL  # cpuinfo uses "key\tvalue", not "key\t: value"
